add_library(shapes_scene STATIC scene.c)
target_include_directories(shapes_scene PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(shapes_scene PUBLIC cydui)
set_target_properties(shapes_scene PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
target_compile_options(shapes_scene PRIVATE ${CYDUI_STRICT_FLAGS})

add_executable(shapes main.c)
target_link_libraries(shapes PRIVATE shapes_scene cydui_mem)
set_target_properties(shapes PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
target_compile_options(shapes PRIVATE ${CYDUI_STRICT_FLAGS})
if(TARGET cydui_sdl)
    target_link_libraries(shapes PRIVATE cydui_sdl)
    target_compile_definitions(shapes PRIVATE CYDUI_HAVE_SDL)
endif()
